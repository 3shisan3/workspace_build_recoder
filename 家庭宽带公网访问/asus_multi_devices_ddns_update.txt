#!/bin/sh
# 阿里云 IPv6 DDNS 多设备版
# 特点：读取配置文件支持多设备，IP未变化不记录日志

# ========== 主配置 ==========
AccessKeyId=""
AccessKeySecret=""
DomainName=""
# 配置文件格式: 设备描述|MAC地址|子域名
CONFIG_FILE="/jffs/scripts/aliddns_devices.conf"
# ============================

# 全局配置
LOG_FILE="/tmp/aliddns_multi.log"
IP_CACHE_DIR="/tmp/aliddns_cache"

# 创建缓存目录
mkdir -p "$IP_CACHE_DIR"

# 精简日志函数
log() {
    local msg="$1"
    # 只记录关键信息
    case "$msg" in
        *❌*|*错误*|*失败*|*✅*|*成功*|*🔄*|*变化*|*创建*|*更新*|*开始*|*结束*|*调试*|*查询*|*找到*|*不存在*)
            # 限制日志大小（保留最后100行）
            if [ -f "$LOG_FILE" ]; then
                local line_count=$(wc -l < "$LOG_FILE" 2>/dev/null | tr -d ' ' || echo 0)
                if [ "$line_count" -gt 100 ]; then
                    tail -n 50 "$LOG_FILE" > "${LOG_FILE}.tmp" 2>/dev/null
                    [ -f "${LOG_FILE}.tmp" ] && mv "${LOG_FILE}.tmp" "$LOG_FILE"
                fi
            fi
            echo "$(date '+%Y-%m-%d %H:%M:%S') - $msg" >> "$LOG_FILE" 2>/dev/null
            ;;
    esac
}

# URL 编码函数 - 修复版
url_encode() {
    local string="$1"
    local length="${#string}"
    local encoded=""
    local i=0
    
    while [ "$i" -lt "$length" ]; do
        local c="${string:$i:1}"
        case "$c" in
            [a-zA-Z0-9.~_-])
                encoded="$encoded$c"
                ;;
            *)
                # 对特殊字符进行编码
                encoded="$encoded$(printf '%%%02X' "'$c")"
                ;;
        esac
        i=$((i + 1))
    done
    
    echo "$encoded"
}

# 清理旧日志
clean_old_logs() {
    [ -f "$LOG_FILE" ] && [ $(find "$LOG_FILE" -mtime +7 2>/dev/null | wc -l) -eq 1 ] && rm -f "$LOG_FILE"
    find "$IP_CACHE_DIR" -name "*.cache" -mtime +1 -delete 2>/dev/null
}

# 获取设备IPv6地址
get_device_ipv6() {
    local description="$1"
    local mac="$2"
    local current_ip=""
    
    if [ -z "$mac" ]; then
        # 路由器自身 - 重点修改这里
        # 方法1: 使用 ip 命令，精确筛选 scope global 地址
        current_ip=$(ip -6 addr show dev $(nvram get wan0_ifname) 2>/dev/null | awk '/inet6.*scope global/ && !/fe80:/ {print $2}' | cut -d'/' -f1 | head -1)
        
        # 方法2: 如果方法1失败，尝试从其他接口获取
        if [ -z "$current_ip" ]; then
            # 尝试常见的 WAN 接口名
            for iface in ppp0 eth0 eth1 eth2 eth3; do
                current_ip=$(ip -6 addr show dev $iface 2>/dev/null | awk '/inet6.*scope global/ && !/fe80:/ {print $2}' | cut -d'/' -f1 | head -1)
                [ -n "$current_ip" ] && break
            done
        fi
    else
        # 内网设备 - 保持原有逻辑，但可增加日志方便调试
        current_ip=$(ip -6 neigh show 2>/dev/null | grep -i "$mac" | grep -v '^fe80:' | awk '{print $1}' | head -1)
        if [ -z "$current_ip" ] && [ -f "/var/lib/misc/dnsmasq.leases" ]; then
            current_ip=$(cat /var/lib/misc/dnsmasq.leases 2>/dev/null | grep -i "$mac" | awk '{print $3}' | head -1)
        fi
        # 可选: 记录获取到的地址类型用于调试
        if [ -n "$current_ip" ]; then
            if echo "$current_ip" | grep -q '^fe80:'; then
                log "⚠️ 警告: 为设备 $description 获取到本地链接地址: $current_ip"
            fi
        fi
    fi
    
    echo "$current_ip"
}

# 阿里云API调用 - 使用单设备脚本的签名逻辑
call_ali_api_original() {
    local action="$1"
    local subdomain="$2"
    local ipv6="$3"
    local record_id="$4"
    
    # 生成时间戳和随机数
    timestamp=$(date -u +%Y-%m-%dT%H:%M:%SZ)
    nonce=$(head -c 16 /dev/urandom 2>/dev/null | openssl base64 2>/dev/null | tr -d '/+=\n' | head -c 16 2>/dev/null)
    [ -z "$nonce" ] && nonce=$(openssl rand -hex 8 2>/dev/null)
    
    # 构造参数 - 完全按照单设备脚本的顺序
    params=""
    
    if [ "$action" = "DescribeDomainRecords" ]; then
        params="AccessKeyId=$(url_encode "$AccessKeyId")&Action=$(url_encode "DescribeDomainRecords")&DomainName=$(url_encode "$DomainName")&Format=$(url_encode "JSON")&RRKeyWord=$(url_encode "$subdomain")&SignatureMethod=$(url_encode "HMAC-SHA1")&SignatureNonce=$(url_encode "$nonce")&SignatureVersion=$(url_encode "1.0")&Timestamp=$(url_encode "$timestamp")&Type=$(url_encode "AAAA")&Version=$(url_encode "2015-01-09")"
    elif [ "$action" = "AddDomainRecord" ]; then
        params="AccessKeyId=$(url_encode "$AccessKeyId")&Action=$(url_encode "AddDomainRecord")&DomainName=$(url_encode "$DomainName")&Format=$(url_encode "JSON")&RR=$(url_encode "$subdomain")&SignatureMethod=$(url_encode "HMAC-SHA1")&SignatureNonce=$(url_encode "$nonce")&SignatureVersion=$(url_encode "1.0")&Timestamp=$(url_encode "$timestamp")&Type=$(url_encode "AAAA")&Value=$(url_encode "$ipv6")&Version=$(url_encode "2015-01-09")"
    elif [ "$action" = "UpdateDomainRecord" ]; then
        params="AccessKeyId=$(url_encode "$AccessKeyId")&Action=$(url_encode "UpdateDomainRecord")&Format=$(url_encode "JSON")&RR=$(url_encode "$subdomain")&RecordId=$(url_encode "$record_id")&SignatureMethod=$(url_encode "HMAC-SHA1")&SignatureNonce=$(url_encode "$nonce")&SignatureVersion=$(url_encode "1.0")&Timestamp=$(url_encode "$timestamp")&Type=$(url_encode "AAAA")&Value=$(url_encode "$ipv6")&Version=$(url_encode "2015-01-09")"
    fi
    
    # 计算签名
    string_to_sign="GET&%2F&$(url_encode "$params")"
    signature=$(echo -n "$string_to_sign" | openssl sha1 -hmac "${AccessKeySecret}&" -binary 2>/dev/null | openssl base64 2>/dev/null | tr -d '\n')
    encoded_signature=$(url_encode "$signature")
    
    # 调用API
    full_url="https://alidns.aliyuncs.com/?${params}&Signature=${encoded_signature}"
    response=$(curl -s -k --connect-timeout 10 "$full_url" 2>/dev/null)
    echo "$response"
}

# 处理单个设备
process_device() {
    local description="$1"
    local mac="$2"
    local subdomain="$3"
    
    local cache_file="$IP_CACHE_DIR/${subdomain//./_}.cache"
    local last_ip=""
    
    log "处理: $description ($subdomain)"
    
    # 获取IPv6地址
    current_ip=$(get_device_ipv6 "$description" "$mac")
    
    if [ -z "$current_ip" ]; then
        log "⚠️ 无法获取IPv6地址: $description"
        return 1
    fi
    
    # 检查IP是否变化
    if [ -f "$cache_file" ]; then
        last_ip=$(cat "$cache_file" 2>/dev/null)
        if [ "$current_ip" = "$last_ip" ] && [ -n "$last_ip" ]; then
            # IP未变化，不记录日志直接退出
            return 0
        fi
        log "🔄 IP地址变化: $last_ip -> $current_ip"
    else
        log "📡 首次获取IP: $current_ip"
    fi
    
    # 查询现有记录 - 使用原脚本的API调用
    log "查询阿里云DNS记录..."
    query_response=$(call_ali_api_original "DescribeDomainRecords" "$subdomain" "" "")
    
    if [ -z "$query_response" ]; then
        log "❌ 查询API调用失败"
        return 1
    fi
    
    # 解析查询结果
    record_id=""
    existing_ip=""
    action=""
    
    if echo "$query_response" | grep -q '"TotalCount"'; then
        total_count=$(echo "$query_response" | grep -o '"TotalCount":[0-9]*' | cut -d: -f2 2>/dev/null | head -1)
        
        if [ -n "$total_count" ] && [ "$total_count" -gt 0 ]; then
            # 提取现有记录
            record_id=$(echo "$query_response" | grep -o '"RecordId":"[^"]*"' | head -1 | cut -d'"' -f4 2>/dev/null)
            existing_ip=$(echo "$query_response" | grep -o '"Value":"[^"]*"' | head -1 | cut -d'"' -f4 2>/dev/null)
            
            if [ -n "$record_id" ] && [ -n "$existing_ip" ]; then
                log "✅ 找到记录: ID=$record_id, IP=$existing_ip"
                
                if [ "$existing_ip" = "$current_ip" ]; then
                    log "IP相同，更新缓存后退出"
                    echo "$current_ip" > "$cache_file"
                    return 0
                fi
                
                # 需要更新
                log "IP不同，准备更新"
                action="UpdateDomainRecord"
            else
                log "❌ 无法解析记录信息"
                action="AddDomainRecord"
            fi
        else
            # 需要创建
            log "➕ 记录不存在，创建新记录"
            action="AddDomainRecord"
        fi
    else
        error_code=$(echo "$query_response" | grep -o '"Code":"[^"]*"' | cut -d'"' -f4 2>/dev/null | head -1)
        error_msg=$(echo "$query_response" | grep -o '"Message":"[^"]*"' | cut -d'"' -f4 2>/dev/null | head -1)
        
        if [ -n "$error_code" ]; then
            log "❌ 查询失败: $error_code - $error_msg"
        else
            log "❌ 查询失败: 未知错误"
        fi
        return 1
    fi
    
    # 执行创建/更新操作
    log "执行操作: $action"
    
    if [ "$action" = "AddDomainRecord" ]; then
        response=$(call_ali_api_original "AddDomainRecord" "$subdomain" "$current_ip" "")
    else
        response=$(call_ali_api_original "UpdateDomainRecord" "$subdomain" "$current_ip" "$record_id")
    fi
    
    if [ -n "$response" ] && echo "$response" | grep -q '"RecordId"'; then
        new_record_id=$(echo "$response" | grep -o '"RecordId":"[^"]*"' | cut -d'"' -f4 2>/dev/null | head -1)
        if [ "$action" = "AddDomainRecord" ]; then
            log "✅ 记录创建成功! RecordId: $new_record_id"
        else
            log "✅ 记录更新成功! RecordId: $new_record_id"
        fi
        # 更新缓存
        echo "$current_ip" > "$cache_file"
    else
        error_code=$(echo "$response" | grep -o '"Code":"[^"]*"' | cut -d'"' -f4 2>/dev/null | head -1)
        error_msg=$(echo "$response" | grep -o '"Message":"[^"]*"' | cut -d'"' -f4 2>/dev/null | head -1)
        
        if [ -n "$error_code" ]; then
            log "❌ 操作失败: $error_code - $error_msg"
            
            # 特殊错误处理
            case "$error_code" in
                "InvalidAccessKeyId.NotFound")
                    log "💡 请检查AccessKeyId是否正确"
                    ;;
                "SignatureDoesNotMatch")
                    log "💡 签名错误，请检查AccessKeySecret"
                    ;;
                "Domain.NotExist")
                    log "💡 域名不存在: $DomainName"
                    ;;
            esac
        else
            log "❌ 操作失败: 未知响应 - $response"
        fi
        return 1
    fi
    
    return 0
}

# 解析配置行
parse_config_line() {
    local line="$1"
    
    # 使用cut解析，但先检查是否有足够的字段
    local description=$(echo "$line" | cut -d'|' -f1)
    local mac=$(echo "$line" | cut -d'|' -f2)
    local subdomain=$(echo "$line" | cut -d'|' -f3)
    
    # 如果没有第三个字段，说明格式不对
    if [ "$subdomain" = "$line" ]; then
        # 可能是只有两个字段（路由器）
        subdomain="$mac"
        mac=""
    fi
    
    # 清理空格
    description=$(echo "$description" | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
    mac=$(echo "$mac" | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
    subdomain=$(echo "$subdomain" | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
    
    echo "$description|$mac|$subdomain"
}

# 主程序
main() {
    clean_old_logs
    log "===== DDNS 多设备检查开始 ====="
    
    # 检查配置文件
    if [ ! -f "$CONFIG_FILE" ]; then
        log "❌ 配置文件不存在: $CONFIG_FILE"
        exit 1
    fi
    
    # 读取配置文件
    success_count=0
    fail_count=0
    
    while IFS= read -r raw_line; do
        # 跳过注释和空行
        [ -z "$raw_line" ] && continue
        echo "$raw_line" | grep -q '^#' && continue
        
        # 解析配置行
        parsed=$(parse_config_line "$raw_line")
        description=$(echo "$parsed" | cut -d'|' -f1)
        mac=$(echo "$parsed" | cut -d'|' -f2)
        subdomain=$(echo "$parsed" | cut -d'|' -f3)
        
        if [ -z "$description" ] || [ -z "$subdomain" ]; then
            log "⚠️ 跳过无效配置行: $raw_line"
            continue
        fi
        
        # 处理设备
        if process_device "$description" "$mac" "$subdomain"; then
            success_count=$((success_count + 1))
        else
            fail_count=$((fail_count + 1))
        fi
        
        # 避免API频率限制
        sleep 2
    done < "$CONFIG_FILE"
    
    log "===== DDNS 检查结束 ====="
    log "✅ 成功: $success_count 个设备"
    [ "$fail_count" -gt 0 ] && log "❌ 失败: $fail_count 个设备"
}

# 工具函数
case "${1:-}" in
    "list")
        echo "=== 设备列表 ==="
        echo ""
        if [ -f "$CONFIG_FILE" ]; then
            echo "描述         MAC地址               子域名"
            echo "-----------------------------------------------"
            while IFS= read -r line; do
                [ -z "$line" ] && continue
                echo "$line" | grep -q '^#' && continue
                
                parsed=$(parse_config_line "$line")
                description=$(echo "$parsed" | cut -d'|' -f1)
                mac=$(echo "$parsed" | cut -d'|' -f2)
                subdomain=$(echo "$parsed" | cut -d'|' -f3)
                
                printf "%-10s %-20s %-15s\n" "$description" "$mac" "$subdomain"
            done < "$CONFIG_FILE"
        else
            echo "配置文件不存在"
        fi
        ;;
    "test")
        if [ -n "$2" ]; then
            # 测试单个设备
            if [ -f "$CONFIG_FILE" ]; then
                while IFS= read -r line; do
                    [ -z "$line" ] && continue
                    echo "$line" | grep -q '^#' && continue
                    
                    parsed=$(parse_config_line "$line")
                    subdomain=$(echo "$parsed" | cut -d'|' -f3)
                    
                    if [ "$subdomain" = "$2" ]; then
                        description=$(echo "$parsed" | cut -d'|' -f1)
                        mac=$(echo "$parsed" | cut -d'|' -f2)
                        
                        echo "测试设备: $description ($subdomain)"
                        process_device "$description" "$mac" "$subdomain"
                        exit
                    fi
                done < "$CONFIG_FILE"
                echo "未找到子域名: $2"
            else
                echo "配置文件不存在"
            fi
        else
            echo "使用方法: $0 test <子域名>"
        fi
        ;;
    "log")
        if [ -f "$LOG_FILE" ]; then
            tail -20 "$LOG_FILE"
        else
            echo "日志文件不存在"
        fi
        ;;
    "config")
        if [ -f "$CONFIG_FILE" ]; then
            cat "$CONFIG_FILE"
        else
            echo "配置文件不存在"
        fi
        ;;
    *)
        main
        ;;
esac
